

az provider register --namespace Microsoft.ContainerService --wait
az provider register --namespace Microsoft.ManagedIdentity --wait
az provider register --namespace Microsoft.Compute --wait


az group create --name RG --location westus2

az aks create --resource-group RG --name AKS --node-count 1 --node-vm-size Standard_D4s_v5 --enable-oidc-issuer --enable-workload-identity --no-ssh-key
#OR
az aks create --resource-group RG --name AKS --node-count 1 --node-vm-size Standard_D4s_v5 --enable-cluster-autoscaler --min-count 1 --max-count 3 --enable-oidc-issuer --enable-workload-identity --no-ssh-key

#OR
az aks update --resource-group RG --name AKS --enable-oidc-issuer --enable-workload-identity

az identity create --resource-group RG --name MI1

az role assignment create --assignee-object-id "$(az identity show --resource-group RG --name MI1 --query principalId --output tsv)" --assignee-principal-type ServicePrincipal --role Contributor --scope "$(az group show --name RG --query id --output tsv)"

#use MI & FIC
az aks get-credentials --resource-group RG --name AKS

kubectl get ns

kubectl create serviceaccount workload-sa --namespace default

kubectl annotate serviceaccount workload-sa --namespace default azure.workload.identity/client-id="$(az identity show --resource-group RG --name MI1 --query clientId --output tsv)" --overwrite

kubectl wait --namespace default --for=condition=Ready pod/wi-test --timeout=180s
